cmake_minimum_required(VERSION 3.14)
project(SimpleChimeraCpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)

set(INC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/include)
set(SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/src)

# --- Controller library (no Bullet) ---
add_library(simplechimera STATIC
  ${SRC_DIR}/two_motor_snn.cpp
  ${SRC_DIR}/kilowatt_chain.cpp
)
target_include_directories(simplechimera PUBLIC ${INC_DIR})

# --- Bullet (simulation) ---
# We only need core libs (Dynamics, Collision, LinearMath). Do not link BulletSoftBody.
find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
  pkg_check_modules(BULLET_PC QUIET bullet)
endif()
if(BULLET_PC_FOUND)
  set(BULLET_INCLUDE_DIRS ${BULLET_PC_INCLUDE_DIRS})
  # Find only the three core libs so we don't pull in BulletSoftBody (often missing from distro)
  find_library(BULLET_DYNAMICS_LIB NAMES BulletDynamics
    HINTS ${BULLET_PC_LIBRARY_DIRS} /opt/homebrew/lib /usr/local/lib)
  find_library(BULLET_COLLISION_LIB NAMES BulletCollision
    HINTS ${BULLET_PC_LIBRARY_DIRS} /opt/homebrew/lib /usr/local/lib)
  find_library(BULLET_LINEARMATH_LIB NAMES LinearMath
    HINTS ${BULLET_PC_LIBRARY_DIRS} /opt/homebrew/lib /usr/local/lib)
  if(BULLET_DYNAMICS_LIB AND BULLET_COLLISION_LIB AND BULLET_LINEARMATH_LIB)
    set(BULLET_LIBRARIES ${BULLET_DYNAMICS_LIB} ${BULLET_COLLISION_LIB} ${BULLET_LINEARMATH_LIB})
  endif()
endif()
if(NOT BULLET_LIBRARIES)
  find_path(BULLET_INCLUDE_DIR NAMES btBulletDynamicsCommon.h
    HINTS $ENV{BULLET_ROOT}/src $ENV{CMAKE_PREFIX_PATH}/include /opt/homebrew/include /usr/local/include)
  find_library(BULLET_DYNAMICS_LIB NAMES BulletDynamics
    HINTS $ENV{BULLET_ROOT}/lib $ENV{CMAKE_PREFIX_PATH}/lib /opt/homebrew/lib /usr/local/lib)
  get_filename_component(BULLET_LIB_DIR ${BULLET_DYNAMICS_LIB} DIRECTORY)
  find_library(BULLET_COLLISION_LIB NAMES BulletCollision HINTS ${BULLET_LIB_DIR})
  find_library(BULLET_LINEARMATH_LIB NAMES LinearMath HINTS ${BULLET_LIB_DIR})
  if(BULLET_INCLUDE_DIR AND BULLET_DYNAMICS_LIB AND BULLET_COLLISION_LIB AND BULLET_LINEARMATH_LIB)
    set(BULLET_INCLUDE_DIRS ${BULLET_INCLUDE_DIR})
    set(BULLET_LIBRARIES ${BULLET_DYNAMICS_LIB} ${BULLET_COLLISION_LIB} ${BULLET_LINEARMATH_LIB})
  endif()
endif()

# --- Simulation library (Bullet + balance env) ---
if(BULLET_LIBRARIES)
  add_library(simplechimera_sim STATIC
    ${SRC_DIR}/bullet_simulation.cpp
    ${SRC_DIR}/balance_env.cpp
  )
  target_include_directories(simplechimera_sim PUBLIC ${INC_DIR})
  target_include_directories(simplechimera_sim PRIVATE ${BULLET_INCLUDE_DIRS})
  target_link_libraries(simplechimera_sim PRIVATE ${BULLET_LIBRARIES})

  # Optional GUI (GLFW + OpenGL)
  find_package(glfw3 QUIET)
  if(NOT glfw3_FOUND)
    find_path(GLFW_INCLUDE_DIR NAMES GLFW/glfw3.h HINTS /opt/homebrew/include /usr/local/include)
    find_library(GLFW_LIBRARY NAMES glfw HINTS /opt/homebrew/lib /usr/local/lib)
    if(GLFW_INCLUDE_DIR AND GLFW_LIBRARY)
      set(GLFW_FOUND TRUE)
    endif()
  endif()

  set(RUN_SIMULATION_SOURCES ${SRC_DIR}/run_simulation.cpp)
  set(RUN_SIMULATION_LIBS simplechimera simplechimera_sim)
  if(glfw3_FOUND OR GLFW_FOUND)
    list(APPEND RUN_SIMULATION_SOURCES ${SRC_DIR}/gui_viewer.cpp)
    if(glfw3_FOUND)
      list(APPEND RUN_SIMULATION_LIBS glfw)
    else()
      list(APPEND RUN_SIMULATION_LIBS ${GLFW_LIBRARY})
    endif()
    if(APPLE)
      list(APPEND RUN_SIMULATION_LIBS "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
    else()
      find_package(OpenGL QUIET)
      if(OpenGL_FOUND)
        list(APPEND RUN_SIMULATION_LIBS OpenGL::GL)
      else()
        list(APPEND RUN_SIMULATION_LIBS GL)
      endif()
    endif()
    message(STATUS "GLFW found: run_simulation will support --gui")
  else()
    message(STATUS "GLFW not found: install with 'brew install glfw' for --gui")
  endif()

  # Full C++ run (replaces Python run.py)
  add_executable(run_simulation ${RUN_SIMULATION_SOURCES})
  target_include_directories(run_simulation PRIVATE ${INC_DIR} ${BULLET_INCLUDE_DIRS})
  if(GLFW_FOUND AND NOT glfw3_FOUND)
    target_include_directories(run_simulation PRIVATE ${GLFW_INCLUDE_DIR})
  endif()
  target_link_libraries(run_simulation PRIVATE ${RUN_SIMULATION_LIBS})
  if(glfw3_FOUND OR GLFW_FOUND)
    target_compile_definitions(run_simulation PRIVATE HAVE_GLFW)
  endif()
  message(STATUS "Bullet found: run_simulation will be built")
else()
  message(STATUS "Bullet not found: install with 'brew install bullet' (macOS) to build run_simulation")
endif()

# --- Standalone controller (no Bullet) ---
add_executable(kilowatt_standalone ${SRC_DIR}/main.cpp)
target_link_libraries(kilowatt_standalone PRIVATE simplechimera)

# --- Optional: MuJoCo simulation (same Kilowatt controller, MuJoCo physics + viewer) ---
# MUJOCO_PATH: directory with include/ and libmujoco*.dylib (pip: site-packages/mujoco). Auto-detected from Python if unset.
set(MUJOCO_PATH "$ENV{MUJOCO_PATH}" CACHE PATH "Path to MuJoCo install (include/ and libmujoco dylib)")
if(NOT MUJOCO_PATH AND DEFINED ENV{MUJOCO_PATH})
  set(MUJOCO_PATH "$ENV{MUJOCO_PATH}")
endif()
# Ignore placeholder or invalid path so auto-detect can run
if(MUJOCO_PATH AND (MUJOCO_PATH STREQUAL "/path/to/mujoco" OR NOT EXISTS "${MUJOCO_PATH}"))
  set(MUJOCO_PATH "" CACHE PATH "" FORCE)
endif()
if(NOT MUJOCO_PATH)
  execute_process(COMMAND python3 -c "import mujoco; import os; print(os.path.dirname(mujoco.__file__))"
    OUTPUT_VARIABLE _mujoco_py OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  if(_mujoco_py AND EXISTS "${_mujoco_py}/include/mujoco/mujoco.h")
    set(MUJOCO_PATH "${_mujoco_py}" CACHE PATH "Path to MuJoCo (from Python)" FORCE)
  endif()
endif()
if(MUJOCO_PATH)
  set(MUJOCO_INC "${MUJOCO_PATH}/include")
  if(NOT EXISTS "${MUJOCO_INC}/mujoco/mujoco.h")
    set(MUJOCO_INC "${MUJOCO_PATH}/mujoco/include")
  endif()
  find_path(MUJOCO_INCLUDE_DIR NAMES mujoco/mujoco.h HINTS "${MUJOCO_INC}" "${MUJOCO_PATH}")
  find_library(MUJOCO_LIBRARY NAMES mujoco mujoco.dylib libmujoco.dylib libmujoco.3.2.0.dylib libmujoco.3.1.0.dylib libmujoco.3.0.0.dylib
    HINTS "${MUJOCO_PATH}/lib" "${MUJOCO_PATH}/bin" "${MUJOCO_PATH}")
  if(NOT MUJOCO_LIBRARY AND MUJOCO_PATH)
    file(GLOB _mujoco_dylib "${MUJOCO_PATH}/libmujoco*.dylib")
    if(_mujoco_dylib)
      list(GET _mujoco_dylib 0 MUJOCO_LIBRARY)
    endif()
  endif()
  if(MUJOCO_INCLUDE_DIR AND MUJOCO_LIBRARY)
    add_executable(run_mujoco ${SRC_DIR}/run_mujoco.cpp)
    target_include_directories(run_mujoco PRIVATE ${INC_DIR} ${MUJOCO_INCLUDE_DIR})
    target_link_libraries(run_mujoco PRIVATE simplechimera ${MUJOCO_LIBRARY})
    if(APPLE)
      target_link_libraries(run_mujoco PRIVATE "-framework OpenGL" "-framework Cocoa" "-framework IOKit" "-framework CoreVideo")
    else()
      find_package(OpenGL QUIET)
      if(OpenGL_FOUND)
        target_link_libraries(run_mujoco PRIVATE OpenGL::GL)
      else()
        target_link_libraries(run_mujoco PRIVATE GL)
      endif()
    endif()
    find_package(glfw3 QUIET)
    if(glfw3_FOUND)
      target_link_libraries(run_mujoco PRIVATE glfw)
    else()
      find_library(GLFW_LIB NAMES glfw HINTS /opt/homebrew/lib /usr/local/lib)
      if(GLFW_LIB)
        target_link_libraries(run_mujoco PRIVATE ${GLFW_LIB})
      endif()
    endif()
    # Pip-installed mujoco dylib has install_name @rpath/mujoco.framework/... which doesn't exist; fix so loader finds the dylib.
    if(APPLE AND MUJOCO_LIBRARY)
      get_filename_component(MUJOCO_LIBRARY_ABS "${MUJOCO_LIBRARY}" ABSOLUTE)
      execute_process(COMMAND sh -c "otool -D '${MUJOCO_LIBRARY_ABS}' 2>/dev/null | tail -1" OUTPUT_VARIABLE _mujoco_install_name OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
      if(_mujoco_install_name)
        add_custom_command(TARGET run_mujoco POST_BUILD
          COMMAND install_name_tool -change "${_mujoco_install_name}" "${MUJOCO_LIBRARY_ABS}" "$<TARGET_FILE:run_mujoco>"
          COMMENT "Fix run_mujoco dylib path for pip-installed mujoco")
      endif()
    endif()
    message(STATUS "MuJoCo found: run_mujoco will be built")
  else()
    message(STATUS "MuJoCo path given but include/lib not found; run_mujoco disabled")
  endif()
else()
  message(STATUS "MuJoCo not set (set MUJOCO_PATH to build run_mujoco)")
endif()

# Optional: build Python bindings if pybind11 is found
find_package(pybind11 CONFIG QUIET)
if(pybind11_FOUND)
  pybind11_add_module(kilowatt_cpp ${SRC_DIR}/pybind_module.cpp)
  target_link_libraries(kilowatt_cpp PRIVATE simplechimera)
endif()
